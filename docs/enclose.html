 32def f(T, L, a=0.5, th=1.0):
 33    r'''Production technology 
 34       $$f(T, L) = \theta \cdot T^{1-\alpha}L^{\alpha}$$ 
 35       '''
 36    return th * T**(1-a) * L**a

 39def mple(te, le, a=0.5, th=1.0, tlbar=DEFAULT_TBAR/DEFAULT_LBAR):
 40    r'''Marginal product of Labor on enclosed land can be written
 41       $$MPL(t_e, l_e) = \alpha \cdot \frac{f(t_e, l_e)}{l_e} \bar l^\alpha$$ 
 42       Since for a Cobb Douglas, $$MPL = \alpha \cdot APL$$.'''
 43    return a * f(te, le, a, th) / le * tlbar**(1-a)

 46def aple(te, le, a=0.5, th=1.0, tlbar=DEFAULT_TBAR/DEFAULT_LBAR):
 47    r'''Average product of Labor 
 48    $$APL(t_e, l_e) = \frac{f(T_e, L_e)}{L_e} =  \frac{f(t_e, l_e)}{l_e} \cdot \bar t^{1-\alpha}$$ 
 49    '''
 50    return f(te, le, a, th) / le * tlbar**(1-a)

 53def mpt(te, le, a=0.5, th=1.0, tlbar=DEFAULT_TBAR/DEFAULT_LBAR):
 54    '''Marginal product of Land on enclosed land'''
 55    return (1-a) * f(te, le, a, th) / te * tlbar**(-a)

 69def Lambda(th, alp, mu):
 70    ''' Key term in expressions. Can return either private or planners 
 71    Lambda:
 72    $$\\mu = 0 \\rightarrow \\Lambda = (\\alpha \\theta)^\\frac{1}{1-\\alpha}$$
 73    $$\\mu = 1 \\rightarrow \\Lambda_o = \\theta^\\frac{1}{1-\\alpha}$$
 74    '''
 75    return ((alp * th) / (1 - mu + alp * mu))**(1 / (1 - alp))

 78def req(te, th=1.0, alp=0.5, ltbar=1.0, mu=0.0):
 79    r'''Decentralized Equilibrium rental given t_e
 80       $$r(t_e) =  \theta f_T(t_e, l_e(t_e)) \cdot \bar l^\alpha$$ 
 81       $$r(t_e) =  \frac{(1-\alpha) \theta  \Lambda_{\mu}^\alpha}{(1+(\Lambda_{\mu}-1)t_e)^\alpha}  \cdot \bar l^\alpha$$ 
 82       
 83    '''
 84    lam = Lambda(th, alp, mu)
 85    return (1-alp) * th * lam**alp * (1+(lam-1)*te)**(-alp) * (ltbar)**(alp)

 88def weq(te, th=1.0, alp=0.5, tlbar=1.0, mu=0.0):
 89    '''Decentralized Equilibrium wage'''
 90    lam = Lambda(th, alp, mu)
 91    return (1+(lam-1)*te)**(1-alp) * (tlbar)**(1-alp)

 94def le(te, th, alp, mu):
 95    '''Private equilibrium labor share on enclosed for given te when 
 96       (1-mu*alp*mu)*APL=MPL  
 97       mu = 0:   APLc = MPLe,   le* in paper
 98       mu = 1:   MPLc = MPLe,   leo in paper
 99       mu in (0,1)   in between partly secure
100       '''
101    lam = Lambda(th, alp, mu)
102    return (lam*te) / (1+(lam-1)*te)

107    return lam / (1+(lam-1)*te)**2

115def totalq(te, th, alp, lbar, mu):
116    '''total output in the economy given te and mu.
117       Note costs of enclosure are not subtracted.'''
118    leq = le(te, th, alp, mu)
119    return (th * f(te, leq, alp, th) + f(1-te, 1-leq, alp, 1)) * lbar**alp

122def z(te, th, alp, lbar):
123    '''output per unit land net of enclosure cost
124       $$z(t_e) = \bar l^\\alpha \\left(1+(\\Lambda_o-1)t_e\\right)^{1-\alpha}$$ '''
125    lam = th**(1/(1-alp))
126    return lbar**alp * (1+(lam-1)*te)**(1-alp)

129def zpv(te, th, alp, lbar):
130    '''output per unit land net of enclosure cost  
131       $$z_d(t_e)= \\bar l^\\alpha \\cdot \\frac{ 1+(\\frac{\\Lambda}{\\alpha}-1)t_e}{(1+(\\Lambda-1)t_e)^\\alpha}$$
132  '''
133    lam = Lambda(th, alp, mu=0)
134    return lbar**alp * (th*te*lam**alp +(1-te)) / (1+(lam-1)*te)**alp

137def zprime(te, th, alp, lbar, mu=1.0):
138    '''derivative of planner's z(t_e) function
139       $$z(t_e) = \\bar l^\\alpha \\cdot (1-\\alpha)(\\Lambda_o -1) \\left(1+(\\Lambda_o-1)t_e \\right)^{-\\alpha}$$
140       '''
141    lam = Lambda(th, alp, mu)
142    return (1-alp) * (lam-1) * lbar**alp * (1+(lam-1)*te)**(-alp)

145def teopt(th, alp, c, lbar):
146    '''Planner enclosure rate. If partial then
147    $$t_e^o =  \\frac{\\bar l}{(\\Lambda_o - 1)} 
148    \\left [   \\frac{(1-\\alpha)(\\Lambda_o - 1)}{c} 
149    \\right ]^\\frac{1}{\\alpha} - \\frac{1}{(\\Lambda_o - 1)}
150    $$'''
151    lam = th**(1/(1-alp))
152    
153    # Using global zprime with mu=1 for planner
154    if zprime(0, th, alp, lbar, mu=1) < c:
155        teopt_val = 0.0
156    elif zprime(1, th, alp, lbar, mu=1) > c:
157        teopt_val = 1.0
158    else:
159        teopt_val = (lbar * (((1-alp)*(lam-1))/c)**(1/alp) - 1) / (lam-1)
160    return teopt_val

163def tepvt(th, alp, c, lbar, mu):
164    '''Private enclosure rate
165        req(te)= rental rate 
166        r(0)<c  : no enclosure 
167        r(1)>c  : full enclosure
168        r(0)>c and r(1)<c : partial enclosure
169           then solve for teopt from foc
170
171        '''
172    thresh = 1/alp - mu * (1-alp)/alp
173    lam = Lambda(th, alp, mu)
174    r0 = req(0, th, alp, lbar, mu)
175    r1 = req(1, th, alp, lbar, mu)
176    if th < thresh:   # r slopes up
177        if r0 >= c:
178            tep = 1.0
179        elif r1 < c:
180            tep = 0.0
181        else:
182            tep = lbar * (lam/(lam-1)) * (th*(1-alp)/c )**(1/alp) - (1/(lam-1))
183    
184    elif th >= thresh:    # r slopes down
185        if r1 >= c:
186            tep = 1.0
187        elif r0 < c:
188            tep = 0.0
189        else:
190            tep = lbar * ( lam/(lam-1)) * (th*(1-alp)/c )**(1/alp) - (1/(lam-1))
191
192    return tep

195def tepvt_g(th, alp, c, lbar, mu):
196    '''Private enclosure rate (global game refinement)
197        just like pvtpart() but adjust for global game
198        If theta < theta_hi then global game refinement says enclose fully if
199        tep (from pvtpart) <= 0.5 otherwise no enclosre.
200        '''
201    thresh = (1 - mu*(1-alp))/alp    
202    tep = tepvt(th, alp, c, lbar, mu=0)
203    
204    tepg = tep
205    if (tep == 1) or (tep == 0):
206        tepg = tep
207    elif (th < thresh):
208        if (tep > 0.5):
209            tepg = 0.0
210        elif (tep <= 0.5):
211            tepg = 1.0
212
213    return tepg

216def dwl(th, alp, c, lbar):
217    '''
218    Returns DWL at each paramter
219    '''
220    teo = teopt(th, alp, c, lbar)
221    # tep = tepvt(th, alp, c, lbar, mu=0)
222    teg = tepvt_g(th, alp, c, lbar, mu=0)
223
224    zo = z(teo, th, alp, lbar) - c*teo
225    zg = zpv(teg, th, alp, lbar) - c*teg
226    return zo, zg

229def dwlpct(th, alp, c, lbar):
230    '''
231    Returns actual/potential at each paramter
232    '''
233    teo = teopt(th, alp, c, lbar)
234    if th > (1/alp):
235        tep = tepvt(th, alp, c, lbar, mu=0)
236    else:
237        tep = tepvt_g(th, alp, c, lbar, mu=0)
238
239    zo = z(teo, th, alp, lbar) - c*teo
240    zg = zpv(tep, th, alp, lbar) - c*tep
241    return zg/zo
